class MoodShots {
    @get:Rule val paparazzi = Paparazzi(deviceConfig = DeviceConfig.PIXEL_6)

    @Test fun moods() = paparazzi.snapshot(name = "moods") {
        val moods = MochiMood.entries.filter { it.emote != null } + listOf(MochiMood.Happy, MochiMood.Surprised, MochiMood.Sleepy)
        Column(Modifier.background(Color(0xFF0B0B10)).padding(12.dp), verticalArrangement = Arrangement.spacedBy(14.dp)) {
            for (row in moods.chunked(4)) {
                Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                    for (m in row) {
                        Column(Modifier.width(88.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                            Row(verticalAlignment = Alignment.Bottom) {
                                MochiGlyph(size = 56.dp, mood = m)
                                MochiGlyph(size = 24.dp, mood = m)
                            }
                            Text(m.name, color = Color.White, fontSize = 11.sp)
                        }
                    }
                }
            }
        }
    }
}
